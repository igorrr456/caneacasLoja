SET NAMES utf8mb4;

DROP DATABASE IF EXISTS loja_canecas_db;
CREATE DATABASE loja_canecas_db CHARACTER SET utf8mb4 COLLATE utf8mb4_general_ci;
USE loja_canecas_db;

-- ---------------------------------------------------------------------
--  TABELAS
-- ---------------------------------------------------------------------

-- Usuarios do PAINEL ADMINISTRATIVO (equipe da loja)
CREATE TABLE usuarios (
    id             INT AUTO_INCREMENT PRIMARY KEY,
    nome           VARCHAR(100) NOT NULL,
    login          VARCHAR(50)  NOT NULL UNIQUE,
    senha_hash     CHAR(64)     NOT NULL,          -- SHA-256 em hexadecimal
    perfil         VARCHAR(20)  NOT NULL,          -- Administrador | Gerente
    data_cadastro  DATETIME     NOT NULL DEFAULT CURRENT_TIMESTAMP,
    status         TINYINT(1)   NOT NULL DEFAULT 1,
    CONSTRAINT ck_usuarios_perfil CHECK (perfil IN ('Administrador','Gerente'))
);

-- Clientes da loja (quem compra; qualquer pessoa pode se cadastrar)
CREATE TABLE clientes (
    id             INT AUTO_INCREMENT PRIMARY KEY,
    nome           VARCHAR(100) NOT NULL,
    cpf            VARCHAR(14)  NOT NULL UNIQUE,
    email          VARCHAR(100) NOT NULL UNIQUE,   -- usado como login
    senha_hash     CHAR(64)     NOT NULL,
    telefone       VARCHAR(20)  NULL,
    cep            VARCHAR(9)   NOT NULL,
    endereco       VARCHAR(150) NOT NULL,
    numero         VARCHAR(10)  NOT NULL,
    complemento    VARCHAR(60)  NULL,
    bairro         VARCHAR(60)  NOT NULL,
    cidade         VARCHAR(60)  NOT NULL,
    uf             CHAR(2)      NOT NULL,
    data_cadastro  DATETIME     NOT NULL DEFAULT CURRENT_TIMESTAMP,
    status         TINYINT(1)   NOT NULL DEFAULT 1
);

CREATE TABLE categorias (
    id             INT AUTO_INCREMENT PRIMARY KEY,
    nome           VARCHAR(60)  NOT NULL UNIQUE,
    descricao      VARCHAR(200) NULL,
    data_cadastro  DATETIME     NOT NULL DEFAULT CURRENT_TIMESTAMP,
    status         TINYINT(1)   NOT NULL DEFAULT 1
);

-- Produtos (as canecas)
CREATE TABLE produtos (
    id             INT AUTO_INCREMENT PRIMARY KEY,
    categoria_id   INT           NOT NULL,
    nome           VARCHAR(100)  NOT NULL,
    descricao      VARCHAR(500)  NULL,
    capacidade_ml  INT           NOT NULL,
    material       VARCHAR(50)   NULL,
    cor            VARCHAR(30)   NULL,
    preco          DECIMAL(10,2) NOT NULL,
    estoque        INT           NOT NULL DEFAULT 0,
    imagem         VARCHAR(200)  NULL,             -- nome do arquivo em wwwroot/img/produtos
    destaque       TINYINT(1)    NOT NULL DEFAULT 0,
    data_cadastro  DATETIME      NOT NULL DEFAULT CURRENT_TIMESTAMP,
    status         TINYINT(1)    NOT NULL DEFAULT 1,
    CONSTRAINT fk_produto_categoria FOREIGN KEY (categoria_id) REFERENCES categorias(id),
    CONSTRAINT ck_produto_preco   CHECK (preco > 0),
    CONSTRAINT ck_produto_estoque CHECK (estoque >= 0)
);

-- Pedidos (CABECALHO)
--   situacao = andamento do pedido (Pendente > Pago > Enviado > Entregue | Cancelado)
--   status   = exclusao logica (1 ativo / 0 excluido)
CREATE TABLE pedidos (
    id                INT AUTO_INCREMENT PRIMARY KEY,
    cliente_id        INT           NOT NULL,
    data_pedido       DATETIME      NOT NULL DEFAULT CURRENT_TIMESTAMP,
    situacao          VARCHAR(20)   NOT NULL DEFAULT 'Pendente',
    forma_pagamento   VARCHAR(30)   NOT NULL,
    endereco_entrega  VARCHAR(300)  NOT NULL,       -- copia do endereco no momento da compra
    valor_produtos    DECIMAL(10,2) NOT NULL DEFAULT 0,
    valor_frete       DECIMAL(10,2) NOT NULL DEFAULT 0,
    valor_total       DECIMAL(10,2) NOT NULL DEFAULT 0,
    status            TINYINT(1)    NOT NULL DEFAULT 1,
    CONSTRAINT fk_pedido_cliente FOREIGN KEY (cliente_id) REFERENCES clientes(id),
    CONSTRAINT ck_pedido_situacao CHECK (situacao IN ('Pendente','Pago','Enviado','Entregue','Cancelado'))
);

-- Itens do pedido (guarda o preco da epoca da compra)
CREATE TABLE pedido_itens (
    id              INT AUTO_INCREMENT PRIMARY KEY,
    pedido_id       INT           NOT NULL,
    produto_id      INT           NOT NULL,
    quantidade      INT           NOT NULL,
    preco_unitario  DECIMAL(10,2) NOT NULL,
    CONSTRAINT fk_item_pedido  FOREIGN KEY (pedido_id)  REFERENCES pedidos(id),
    CONSTRAINT fk_item_produto FOREIGN KEY (produto_id) REFERENCES produtos(id),
    CONSTRAINT ck_item_qtd CHECK (quantidade > 0),
    CONSTRAINT uq_item_produto UNIQUE (pedido_id, produto_id)
);

-- =====================================================================
--  USUARIOS (painel administrativo)
-- =====================================================================
CREATE PROCEDURE sp_usuario_login(IN p_login VARCHAR(50), IN p_senha CHAR(64))
BEGIN
    SELECT id, nome, login, perfil
      FROM usuarios
     WHERE login = p_login AND senha_hash = p_senha AND status = 1;
END$$

CREATE PROCEDURE sp_usuario_listar(IN p_status TINYINT)
BEGIN
    SELECT id, nome, login, perfil, data_cadastro, status
      FROM usuarios
     WHERE status = p_status
     ORDER BY nome;
END$$

CREATE PROCEDURE sp_usuario_obter(IN p_id INT)
BEGIN
    SELECT id, nome, login, perfil, data_cadastro, status
      FROM usuarios
     WHERE id = p_id;
END$$

CREATE PROCEDURE sp_usuario_criar(IN p_nome VARCHAR(100), IN p_login VARCHAR(50),
                                  IN p_senha CHAR(64), IN p_perfil VARCHAR(20))
BEGIN
    IF EXISTS (SELECT 1 FROM usuarios WHERE login = p_login) THEN
        SIGNAL SQLSTATE '45000' SET MESSAGE_TEXT = 'Já existe um usuário com este login.';
    END IF;

    INSERT INTO usuarios (nome, login, senha_hash, perfil)
    VALUES (p_nome, p_login, p_senha, p_perfil);
END$$

-- p_senha NULL ou vazio = mantém a senha atual
CREATE PROCEDURE sp_usuario_editar(IN p_id INT, IN p_nome VARCHAR(100), IN p_login VARCHAR(50),
                                   IN p_perfil VARCHAR(20), IN p_senha CHAR(64))
BEGIN
    IF EXISTS (SELECT 1 FROM usuarios WHERE login = p_login AND id <> p_id) THEN
        SIGNAL SQLSTATE '45000' SET MESSAGE_TEXT = 'Já existe um usuário com este login.';
    END IF;

    -- nao deixa o sistema ficar sem nenhum Administrador ativo
    IF p_perfil <> 'Administrador'
       AND EXISTS (SELECT 1 FROM usuarios WHERE id = p_id AND perfil = 'Administrador' AND status = 1)
       AND (SELECT COUNT(*) FROM usuarios WHERE perfil = 'Administrador' AND status = 1) <= 1 THEN
        SIGNAL SQLSTATE '45000' SET MESSAGE_TEXT = 'O sistema precisa de pelo menos um Administrador ativo.';
    END IF;

    UPDATE usuarios
       SET nome = p_nome,
           login = p_login,
           perfil = p_perfil,
           senha_hash = IF(p_senha IS NULL OR p_senha = '', senha_hash, p_senha)
     WHERE id = p_id;
END$$

CREATE PROCEDURE sp_usuario_excluir(IN p_id INT)
BEGIN
    IF EXISTS (SELECT 1 FROM usuarios WHERE id = p_id AND perfil = 'Administrador' AND status = 1)
       AND (SELECT COUNT(*) FROM usuarios WHERE perfil = 'Administrador' AND status = 1) <= 1 THEN
        SIGNAL SQLSTATE '45000' SET MESSAGE_TEXT = 'O sistema precisa de pelo menos um Administrador ativo.';
    END IF;

    UPDATE usuarios SET status = 0 WHERE id = p_id;
END$$

CREATE PROCEDURE sp_usuario_reativar(IN p_id INT)
BEGIN
    UPDATE usuarios SET status = 1 WHERE id = p_id;
END$$

CREATE PROCEDURE sp_usuario_alterar_senha(IN p_id INT, IN p_senha_atual CHAR(64), IN p_senha_nova CHAR(64))
BEGIN
    IF NOT EXISTS (SELECT 1 FROM usuarios WHERE id = p_id AND senha_hash = p_senha_atual) THEN
        SIGNAL SQLSTATE '45000' SET MESSAGE_TEXT = 'Senha atual incorreta.';
    END IF;

    UPDATE usuarios SET senha_hash = p_senha_nova WHERE id = p_id;
END$$

CREATE PROCEDURE sp_cliente_login(IN p_email VARCHAR(100), IN p_senha CHAR(64))
BEGIN
    SELECT id, nome, email
      FROM clientes
     WHERE email = p_email AND senha_hash = p_senha AND status = 1;
END$$

CREATE PROCEDURE sp_cliente_criar(IN p_nome VARCHAR(100), IN p_cpf VARCHAR(14), IN p_email VARCHAR(100),
                                  IN p_senha CHAR(64), IN p_telefone VARCHAR(20), IN p_cep VARCHAR(9),
                                  IN p_endereco VARCHAR(150), IN p_numero VARCHAR(10),
                                  IN p_complemento VARCHAR(60), IN p_bairro VARCHAR(60),
                                  IN p_cidade VARCHAR(60), IN p_uf CHAR(2))
BEGIN
    IF EXISTS (SELECT 1 FROM clientes WHERE email = p_email) THEN
        SIGNAL SQLSTATE '45000' SET MESSAGE_TEXT = 'Já existe um cadastro com este e-mail.';
    END IF;
    IF EXISTS (SELECT 1 FROM clientes WHERE cpf = p_cpf) THEN
        SIGNAL SQLSTATE '45000' SET MESSAGE_TEXT = 'Já existe um cadastro com este CPF.';
    END IF;

    INSERT INTO clientes (nome, cpf, email, senha_hash, telefone, cep, endereco, numero,
                          complemento, bairro, cidade, uf)
    VALUES (p_nome, p_cpf, p_email, p_senha, p_telefone, p_cep, p_endereco, p_numero,
            p_complemento, p_bairro, p_cidade, p_uf);

    SELECT LAST_INSERT_ID() AS id;
END$$

CREATE PROCEDURE sp_cliente_obter(IN p_id INT)
BEGIN
    SELECT id, nome, cpf, email, telefone, cep, endereco, numero, complemento,
           bairro, cidade, uf, data_cadastro, status
      FROM clientes
     WHERE id = p_id;
END$$

                                   IN p_email VARCHAR(100), IN p_telefone VARCHAR(20),
                                   IN p_cep VARCHAR(9), IN p_endereco VARCHAR(150),

                                   IN p_bairro VARCHAR(60), IN p_cidade VARCHAR(60), IN p_uf CHAR(2))
BEGIN

    UPDATE clientes

           bairro = p_bairro, cidade = p_cidade, uf = p_uf

END$$

-- Listagem do painel: traz tambem quantidade de pedidos e total comprado
CREATE PROCEDURE sp_cliente_listar(IN p_busca VARCHAR(100), IN p_status TINYINT)
BEGIN
    SELECT c.id, c.nome, c.cpf, c.email, c.telefone, c.cidade, c.uf, c.data_cadastro, c.status,
           (SELECT COUNT(*) FROM pedidos p
             WHERE p.cliente_id = c.id AND p.status = 1 AND p.situacao <> 'Cancelado') AS total_pedidos,
           (SELECT IFNULL(SUM(p.valor_total), 0) FROM pedidos p
             WHERE p.cliente_id = c.id AND p.status = 1 AND p.situacao <> 'Cancelado') AS total_comprado
      FROM clientes c
     WHERE c.status = p_status
       AND (p_busca = '' OR c.nome LIKE CONCAT('%', p_busca, '%')
                         OR c.email LIKE CONCAT('%', p_busca, '%')
                         OR c.cpf LIKE CONCAT('%', p_busca, '%'))

-- Vitrine da loja: so produtos ativos de categorias ativas

BEGIN
    SELECT p.id, p.categoria_id, c.nome AS categoria_nome, p.nome, p.descricao, p.capacidade_ml,
           p.material, p.cor, p.preco, p.estoque, p.imagem, p.destaque, p.data_cadastro, p.status
      FROM produtos p
      JOIN categorias c ON c.id = p.categoria_id AND c.status = 1
     WHERE p.status = 1
       AND (p_categoria_id = 0 OR p.categoria_id = p_categoria_id)
       AND (p_busca = '' OR p.nome LIKE CONCAT('%', p_busca, '%')
                         OR p.descricao LIKE CONCAT('%', p_busca, '%')
                         OR p.cor LIKE CONCAT('%', p_busca, '%'))
     ORDER BY
        CASE WHEN p_ordem = 'menor' THEN p.preco END ASC,
        CASE WHEN p_ordem = 'maior' THEN p.preco END DESC,
        CASE WHEN p_ordem = 'novos' THEN p.data_cadastro END DESC,
        p.nome ASC;
END$$

-- =====================================================================
--  PEDIDOS
--  Fluxo da compra (feito em UMA transacao pelo C#):
--    1) sp_pedido_criar            -> cria o cabecalho e devolve o id
--    2) sp_pedido_item_adicionar   -> para cada item: valida e baixa o estoque
--    3) sp_pedido_finalizar        -> soma os itens, calcula frete e total
-- =====================================================================
CREATE PROCEDURE sp_pedido_criar(IN p_cliente_id INT, IN p_forma_pagamento VARCHAR(30),
                                 IN p_endereco VARCHAR(300))
BEGIN
    IF NOT EXISTS (SELECT 1 FROM clientes WHERE id = p_cliente_id AND status = 1) THEN
        SIGNAL SQLSTATE '45000' SET MESSAGE_TEXT = 'Cliente inválido.';
    END IF;

    INSERT INTO pedidos (cliente_id, forma_pagamento, endereco_entrega)
    VALUES (p_cliente_id, p_forma_pagamento, p_endereco);

    SELECT LAST_INSERT_ID() AS id;
END$$

CREATE PROCEDURE sp_pedido_item_adicionar(IN p_pedido_id INT, IN p_produto_id INT, IN p_quantidade INT)
BEGIN
    DECLARE v_preco   DECIMAL(10,2) DEFAULT NULL;
    DECLARE v_estoque INT DEFAULT 0;
    DECLARE v_nome    VARCHAR(100) DEFAULT '';
    DECLARE v_msg     VARCHAR(120);

    IF p_quantidade IS NULL OR p_quantidade <= 0 THEN
        SIGNAL SQLSTATE '45000' SET MESSAGE_TEXT = 'Quantidade inválida.';
    END IF;

    -- trava a linha do produto para que duas compras simultaneas nao vendam o mesmo estoque
    SELECT p.preco, p.estoque, p.nome INTO v_preco, v_estoque, v_nome
      FROM produtos p
      JOIN categorias c ON c.id = p.categoria_id AND c.status = 1
     WHERE p.id = p_produto_id AND p.status = 1
       FOR UPDATE;

    IF v_preco IS NULL THEN
        SIGNAL SQLSTATE '45000' SET MESSAGE_TEXT = 'Um dos produtos do carrinho não está mais disponível.';
    END IF;

    IF v_estoque < p_quantidade THEN
        SET v_msg = CONCAT('Estoque insuficiente para "', LEFT(v_nome, 40), '". Disponível: ', v_estoque);
        SIGNAL SQLSTATE '45000' SET MESSAGE_TEXT = v_msg;
    END IF;

    UPDATE produtos SET estoque = estoque - p_quantidade WHERE id = p_produto_id;

    INSERT INTO pedido_itens (pedido_id, produto_id, quantidade, preco_unitario)
    VALUES (p_pedido_id, p_produto_id, p_quantidade, v_preco);
END$$

-- Regra de frete: gratis a partir de R$ 150,00; abaixo disso R$ 15,90
-- (a mesma regra esta em Models/RegrasLoja.cs, usada para exibir o carrinho)
CREATE PROCEDURE sp_pedido_finalizar(IN p_pedido_id INT)
BEGIN
    DECLARE v_produtos DECIMAL(10,2);
    DECLARE v_frete    DECIMAL(10,2);

    SELECT IFNULL(SUM(quantidade * preco_unitario), 0) INTO v_produtos
      FROM pedido_itens WHERE pedido_id = p_pedido_id;

    IF v_produtos <= 0 THEN
        SIGNAL SQLSTATE '45000' SET MESSAGE_TEXT = 'O pedido não possui itens.';
    END IF;

    SET v_frete = IF(v_produtos >= 150, 0, 15.90);

    UPDATE pedidos
       SET valor_produtos = v_produtos,
           valor_frete    = v_frete,
           valor_total    = v_produtos + v_frete
     WHERE id = p_pedido_id;
END$$

-- Avanca o pedido (Pendente > Pago > Enviado > Entregue) ou cancela (so Pendente/Pago).
-- Ao cancelar, o estoque dos itens volta para a loja.
CREATE PROCEDURE sp_pedido_alterar_situacao(IN p_id INT, IN p_situacao VARCHAR(20))
BEGIN
    DECLARE v_atual VARCHAR(20) DEFAULT NULL;

    SELECT situacao INTO v_atual FROM pedidos WHERE id = p_id AND status = 1 FOR UPDATE;

    IF v_atual IS NULL THEN
        SIGNAL SQLSTATE '45000' SET MESSAGE_TEXT = 'Pedido não encontrado.';
    END IF;

    IF p_situacao NOT IN ('Pendente','Pago','Enviado','Entregue','Cancelado') THEN
        SIGNAL SQLSTATE '45000' SET MESSAGE_TEXT = 'Situação inválida.';
    END IF;

    IF v_atual IN ('Entregue','Cancelado') THEN
        SIGNAL SQLSTATE '45000' SET MESSAGE_TEXT = 'Pedido encerrado: a situação não pode mais ser alterada.';
    END IF;

    IF p_situacao = 'Cancelado' THEN
        IF v_atual NOT IN ('Pendente','Pago') THEN
            SIGNAL SQLSTATE '45000' SET MESSAGE_TEXT = 'Pedido já enviado não pode ser cancelado.';
        END IF;

        UPDATE produtos pr
          JOIN pedido_itens i ON i.produto_id = pr.id
           SET pr.estoque = pr.estoque + i.quantidade
         WHERE i.pedido_id = p_id;
    ELSEIF FIELD(p_situacao, 'Pendente','Pago','Enviado','Entregue') <= FIELD(v_atual, 'Pendente','Pago','Enviado','Entregue') THEN
        SIGNAL SQLSTATE '45000' SET MESSAGE_TEXT = 'A situação do pedido só pode avançar.';
    END IF;

    UPDATE pedidos SET situacao = p_situacao WHERE id = p_id;
END$$

-- O cliente so cancela o proprio pedido, e apenas enquanto estiver Pendente
CREATE PROCEDURE sp_pedido_cancelar_cliente(IN p_id INT, IN p_cliente_id INT)
BEGIN
    IF NOT EXISTS (SELECT 1 FROM pedidos WHERE id = p_id AND cliente_id = p_cliente_id AND status = 1) THEN
        SIGNAL SQLSTATE '45000' SET MESSAGE_TEXT = 'Pedido não encontrado.';
    END IF;

    IF NOT EXISTS (SELECT 1 FROM pedidos WHERE id = p_id AND situacao = 'Pendente') THEN
        SIGNAL SQLSTATE '45000' SET MESSAGE_TEXT = 'Só é possível cancelar pedidos com pagamento pendente.';
    END IF;

    CALL sp_pedido_alterar_situacao(p_id, 'Cancelado');
END$$

-- =====================================================================
--  DASHBOARD (painel administrativo)
--  "Venda" = pedido ativo com situacao Pago, Enviado ou Entregue.
--  p_dias = tamanho do periodo (ex.: 30 = ultimos 30 dias, contando hoje)
-- =====================================================================
CREATE PROCEDURE sp_dashboard_resumo(IN p_dias INT)
BEGIN
    DECLARE v_inicio DATE;
    SET v_inicio = DATE_SUB(CURDATE(), INTERVAL (p_dias - 1) DAY);

    SELECT
        (SELECT IFNULL(SUM(valor_total), 0) FROM pedidos
          WHERE status = 1 AND situacao IN ('Pago','Enviado','Entregue')
            AND DATE(data_pedido) >= v_inicio)                                   AS faturamento,
        (SELECT COUNT(*) FROM pedidos
          WHERE status = 1 AND situacao IN ('Pago','Enviado','Entregue')
            AND DATE(data_pedido) >= v_inicio)                                   AS total_vendas,
        (SELECT IFNULL(SUM(i.quantidade), 0) FROM pedido_itens i
           JOIN pedidos p ON p.id = i.pedido_id
          WHERE p.status = 1 AND p.situacao IN ('Pago','Enviado','Entregue')
            AND DATE(p.data_pedido) >= v_inicio)                                 AS canecas_vendidas,
        (SELECT IFNULL(AVG(valor_total), 0) FROM pedidos
          WHERE status = 1 AND situacao IN ('Pago','Enviado','Entregue')
            AND DATE(data_pedido) >= v_inicio)                                   AS ticket_medio,
        (SELECT COUNT(*) FROM pedidos WHERE status = 1 AND situacao = 'Pendente') AS pedidos_pendentes,
        (SELECT COUNT(*) FROM pedidos WHERE status = 1 AND situacao = 'Pago')     AS pedidos_a_enviar,
        (SELECT COUNT(*) FROM clientes WHERE status = 1)                         AS total_clientes,
        (SELECT COUNT(*) FROM clientes
          WHERE status = 1 AND DATE(data_cadastro) >= v_inicio)                  AS clientes_novos,
        (SELECT COUNT(*) FROM produtos WHERE status = 1)                         AS produtos_ativos,
        (SELECT COUNT(*) FROM produtos WHERE status = 1 AND estoque <= 5)        AS estoque_baixo;
END$$

-- Dias sem venda nao aparecem aqui (o C# completa com zero)
CREATE PROCEDURE sp_dashboard_vendas_por_dia(IN p_dias INT)
BEGIN
    SELECT DATE(data_pedido) AS dia, COUNT(*) AS qtd_pedidos, SUM(valor_total) AS valor
      FROM pedidos
     WHERE status = 1 AND situacao IN ('Pago','Enviado','Entregue')
       AND DATE(data_pedido) >= DATE_SUB(CURDATE(), INTERVAL (p_dias - 1) DAY)
     GROUP BY DATE(data_pedido)
     ORDER BY dia;
END$$

    SELECT pr.id, pr.nome, pr.imagem, SUM(i.quantidade) AS quantidade,
           SUM(i.quantidade * i.preco_unitario) AS valor
      FROM pedido_itens i
      JOIN pedidos p   ON p.id = i.pedido_id
      JOIN produtos pr ON pr.id = i.produto_id
     WHERE p.status = 1 AND p.situacao IN ('Pago','Enviado','Entregue')
       AND DATE(p.data_pedido) >= DATE_SUB(CURDATE(), INTERVAL (p_dias - 1) DAY)
     GROUP BY pr.id, pr.nome, pr.imagem
     ORDER BY quantidade DESC, valor DESC
     LIMIT 5;
